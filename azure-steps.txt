// File : angular-azure-A-oct26/src/app/components/fetch-data-from-api/fetch-data-from-api.ts
import { Component, OnInit, inject, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';

@Component({
  selector: 'app-fetch-data-from-api',
  standalone: true,
  imports: [], // CommonModule is no longer needed here!
  templateUrl: './fetch-data-from-api.html',
  styleUrls: ['./fetch-data-from-api.css']
})
export class FetchDataFromAPI implements OnInit {
  private http = inject(HttpClient);
  
  users = signal<any[]>([]);
  isLoading = signal<boolean>(true);

  ngOnInit(): void {
    this.http.get<any[]>('https://jsonplaceholder.typicode.com/users')
      .subscribe({
        next: (data) => {
          this.users.set(data);
          this.isLoading.set(false);
        },
        error: (err) => {
          console.error('Error fetching data:', err);
          this.isLoading.set(false);
        }
      });
  }
}
